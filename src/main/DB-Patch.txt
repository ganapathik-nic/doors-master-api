CREATE TABLE doors_signing_certificates (
    id SERIAL PRIMARY KEY,
    key_id VARCHAR(50) NOT NULL UNIQUE,       -- e.g., "doors-pki-v1", "doors-pki-v2"
    display_name VARCHAR(100) NOT NULL,       -- e.g., "Initial Core Certificate Set"
    is_active BOOLEAN DEFAULT false,          -- Flags which key pair is actively signing responses
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    activated_at TIMESTAMP,
    expiry_at TIMESTAMP
);
